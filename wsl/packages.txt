# update and upgrade (refresh package list and install updates)
sudo apt update
sudo apt upgrade

# trash-cli (send files to the trash instead of deleting them)
sudo apt install trash-cli

# tree (show directories as a tree)
sudo apt install tree

# bat (cat with syntax highlighting)
sudo apt install bat

# htop (process monitor)
sudo apt install htop

# unzip (extract .zip files)
sudo apt install unzip

# ffmpeg (video and audio tools)
sudo apt install ffmpeg

# poppler-utils (PDF tools like pdftotext)
sudo apt install poppler-utils

# neovim (terminal text editor)
sudo apt install neovim

# chafa (show images in the terminal)
sudo apt install chafa

# node and npm (JavaScript runtime and its package manager)
sudo apt install nodejs npm

# c/c++ build toolchain (gcc, g++ and make)
sudo apt install build-essential

# java (JDK 25: compiler and runtime)
sudo apt install openjdk-25-jdk

# docker (containers and compose)
sudo apt install docker.io docker-compose-v2

# rust (rustup: Rust compiler and cargo installer)
curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh
